Output: 21

Output: Audi

Section 6: Increment / Decrement (33–36)

let a = 5;
a++;
console.log(a);
Output: 6
let b = 5;
let c = b++;
console.log(b, c);
Output: 6 5
let x = 5;
let y = ++x;
console.log(x, y);
Output: 6 6
let m = 3;
let n = m--;
console.log(m, n);
Output: 2 3
